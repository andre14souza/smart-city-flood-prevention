Especificação técnica da arquitetura

Documentação de engenharia para apresentação acadêmica e implantação municipal

1. Camada de borda (Edge)

Sensor HC-SR04 mede lâmina d'água sem contato. ESP32 (ou gêmeo digital Python) processa médias e transmite via MQTT leve a cada 5 segundos.

2. Gateway local (Node-RED)

Autonomia offline garantida. Armazena leituras em contingência, aciona sirenes locais e reenvia para a nuvem quando a conectividade é restaurada.

3. Nuvem AWS (sa-east-1)

IoT Core com TLS, Lambda sem servidor ocioso, DynamoDB para séries temporais e SNS para broadcast de emergência à Defesa Civil.

Fluxo de dados:
  ESP32/Simulador → [MQTT broker] → Node-RED → AWS IoT Core
                                             ↓
                                     Lambda (processamento)
                                             ↓
                              DynamoDB ← registro | SNS → alerta SMS/e-mail
#include <WiFi.h>
#include <PubSubClient.h>
#include <ArduinoJson.h>

const char* ssid        = "WIFI_MUNICIPAL";
const char* password    = "SENHA";
const char* mqtt_server = "192.168.1.100";

const int TRIG_PIN = 5;
const int ECHO_PIN = 18;

WiFiClient   espClient;
PubSubClient client(espClient);

float medirDistancia() {
  digitalWrite(TRIG_PIN, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long dur = pulseIn(ECHO_PIN, HIGH);
  float d = dur * 0.0343 / 2.0;
  return (d > 0 && d < 400) ? d : -1;
}

void loop() {
  if (!client.connected()) reconnect();
  client.loop();

  float nivel = medirDistancia();
  if (nivel != -1) {
    StaticJsonDocument<200> doc;
    doc["sensor_id"]   = "bueiro_01";
    doc["distancia_cm"] = nivel;
    doc["timestamp"]   = millis();

    char buf[256];
    serializeJson(doc, buf);
    client.publish("cidades_inteligentes/enchentes/bueiro_01", buf);
  }
  delay(5000);
}

# Função AWS Lambda — Processador Hidrológico
import boto3, os, time
from decimal import Decimal

dynamodb = boto3.resource('dynamodb')
sns      = boto3.client('sns')

def lambda_handler(event, context):
    sensor_id   = event.get('sensor_id', 'bueiro_01')
    distancia   = float(event.get('distancia_cm', 200.0))
    epoch_now   = int(time.time())

    nivel_cm    = max(0.0, min(200.0, 200.0 - distancia))
    nivel_pct   = round((nivel_cm / 200.0) * 100.0, 1)
    status      = 'CRITICO' if nivel_pct >= 80 else 'NORMAL'

    dynamodb.Table('FloodSensorReadings').put_item(Item={
        'sensor_id'    : sensor_id,
        'timestamp'    : epoch_now,
        'distancia_cm' : Decimal(str(distancia)),
        'nivel_pct'    : Decimal(str(nivel_pct)),
        'status'       : status
    })

    if nivel_pct >= 80:
        sns.publish(
            TopicArn=os.environ['SNS_TOPIC_ARN'],
            Subject =f"ALERTA DEFESA CIVIL: {sensor_id}",
            Message =f"Cota de transbordamento: {nivel_pct}%"
        )

    return {'statusCode': 200, 'nivel_pct': nivel_pct}
